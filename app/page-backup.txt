function speak(text:string) {
  if(typeof window==="undefined" || !("speechSynthesis" in window)) return;
  window.speechSynthesis.cancel();
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang="fr-FR";
  utterance.rate=0.83;
  const voices=window.speechSynthesis.getVoices();
  const french=voices.find(v=>v.lang.toLowerCase().startsWith("fr"));
  if(french) utterance.voice=french;
  window.speechSynthesis.speak(utterance);
}

function normalize(value:string) {
  return value.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g,"").replace(/[^a-z0-9 ]/g,"").trim();
}

function Photo({product,className=""}:{product:Product;className?:string}) {
  const [failed,setFailed]=useState(false);
  return failed
    ? <div className={`flex items-center justify-center bg-[#f3e7ef] text-6xl ${className}`}>{product.emoji}</div>
    : <img src={product.image} alt={product.name} className={`object-cover ${className}`} onError={()=>setFailed(true)}/>;
}

function Dot({color}:{color:string}) {
  return <span className="inline-block h-5 w-5 rounded-full border border-black/15" style={{backgroundColor:colorHex[color]||color}}/>;
}

function newPlayer(id: string, nickname: string): Player {
  return {
    id,
    nickname,
    points: 0,
    completed: 0,
    history: [],
    activeMission: null,
    cart: [],
    earned: 0,
    bonus: 0,
    surpriseUsed: false,
    badges: [],
    listeningBest: 0,
  };
}

function fromRow(row: Record<string, unknown>): Player {
  return {
    id: String(row.id),
    nickname: String(row.nickname ?? "Élève"),
    points: Number(row.points ?? 0),
    completed: Number(row.completed ?? 0),
    history: Array.isArray(row.history)
      ? (row.history as HistoryItem[])
      : [],
    activeMission:
      typeof row.active_mission === "number"
        ? row.active_mission
        : null,
    cart: Array.isArray(row.cart)
      ? (row.cart as CartItem[])
      : [],
    earned: Number(row.earned ?? 0),
    bonus: Number(row.bonus ?? 0),
    surpriseUsed: Boolean(row.surprise_used),
    badges: Array.isArray(row.badges)
      ? (row.badges as string[])
      : [],
    listeningBest: Number(row.listening_best ?? 0),
  };
}

function toRow(p: Player) {
  return {
    id: p.id,
    nickname: p.nickname,
    points: p.points,
    completed: p.completed,
    history: p.history,
    active_mission: p.activeMission,
    cart: p.cart,
    earned: p.earned,
    bonus: p.bonus,
    surprise_used: p.surpriseUsed,
    badges: p.badges,
    listening_best: p.listeningBest,
    updated_at: new Date().toISOString(),
  };
}

  // Save progress in order to avoid older updates

  function queueSave(p: Player) {
    const generation = generationRef.current;
    const revision = ++revisionRef.current;

    setSaving("saving");

    pendingRef.current = pendingRef.current
      .catch(() => {})
      .then(async () => {
        if (generation !== generationRef.current) return;

        const {
          data: { user },
          error: authError,
        } = await supabase.auth.getUser();

        if (
          authError ||
          !user ||
          user.id !== p.id ||
          user.is_anonymous
        ) {
          throw new Error(
            "Session expirée. Reconnecte-toi."
          );
        }

        const { error } = await supabase
          .from("student_progress")
          .upsert(toRow(p), { onConflict: "id" });

        if (error) throw error;

        savedRevisionRef.current = revision;

        if (
          mountedRef.current &&
          generation === generationRef.current &&
          revision === revisionRef.current
        ) {
          setSaving("saved");
        }
      })
      .catch((e) => {
        if (
          mountedRef.current &&
          generation === generationRef.current
        ) {
          setSaving("error");
          setAuthMessage(
            e instanceof Error
              ? `Sauvegarde impossible : ${e.message}`
              : "Sauvegarde impossible."
          );
        }
      });

    const current = playerRef.current;
    if (!current) return;

    const next = fn(current);
    if (next === current) return;

    playerRef.current = next;
    setPlayer(next);
    queueSave(next);
  }

        points: total,

        badges: Array.from(

          password,
          options: {
            data: {
              nickname: nickname.trim().slice(0, 30),
            },

     } else if (data.user) {
  await loadAccount(

  );
}
      } else {
        const { data, error } =
          await supabase.auth.signInWithPassword({

            password,
          });

        if (error) throw error;

        await loadAccount(data.user.id);
      }

      setPassword("");
    } catch (e) {
      setAuthMessage(
        e instanceof Error
          ? e.message
          : "Connexion impossible."
      );
    } finally {
      setAuthBusy(false);
    }
  }

  async function loadAccount(
    userId: string,
    requestedNickname?: string
  ) {
    const { data, error } = await supabase
      .from("student_progress")
      .select("*")
      .eq("id", userId)
      .maybeSingle();

    if (error) throw error;

    let p: Player;

    if (data) {
      p = fromRow(data as Record<string, unknown>);
    } else {
      const {
        data: { user },
      } = await supabase.auth.getUser();

      const name =
        requestedNickname ||
        String(user?.user_metadata?.nickname ?? "")
          .trim()
          .slice(0, 30);

      if (!name) {
        setAuthMessage(
          "Compte connecté. Saisis ton pseudo pour créer ton profil."
        );
        setNickname("");
        return;
      }

      p = newPlayer(userId, name);

      const { error: insertError } = await supabase
        .from("student_progress")
        .insert(toRow(p));

      if (insertError) throw insertError;
    }

    generationRef.current++;
    pendingRef.current = Promise.resolve();
    revisionRef.current = 0;
    savedRevisionRef.current = 0;

    playerRef.current = p;
    setPlayer(p);
    setScreen("home");
    setAuthMessage("");
    setSaving("saved");
  }

      const {
        data: { user },
        error,
      } = await supabase.auth.getUser();

      if (error || !user || user.is_anonymous) {

          "Reconnecte-toi pour créer ton profil."
        );

      await loadAccount(
        user.id,
        nickname.trim().slice(0, 30)
      );
    } catch (e) {
      setAuthMessage(
        e instanceof Error
          ? e.message
          : "Impossible de créer le profil."
      );
    } finally {
      setAuthBusy(false);
    }
  }

    setAuthBusy(true);

    try {
      await pendingRef.current.catch(() => {});

        saving === "error" &&
        !window.confirm(
          "La dernière sauvegarde a échoué. Se déconnecter quand même ?"
        )
      ) {
        return;

      const { error } = await supabase.auth.signOut();

      playerRef.current = null;

      setPlayer(null);
      setSelected(null);
      setScreen("home");
      setAuthMessage("");
      setSaving("saved");
      setEmail("");
      setPassword("");
    } catch (e) {
      setAuthMessage(
        e instanceof Error
          ? e.message
          : "Déconnexion impossible."

    } finally {
      setAuthBusy(false);

  function advance() {
    if (!selected) return;

    const response = normalize(answer);

    if (stage === "greeting") {
      if (
        response.includes("bonjour") ||
        response.includes("bonsoir") ||
        response.includes("salut")
      ) {
        addPoints(5);
        setStage("request");
        setMessage("Bravo ! +5 points");
      } else {
        setMessage(
          "Essaie de saluer le vendeur en français."
        );
        return;
      }
    }

    if (stage === "request") {
      if (
        response.includes("voudrais") ||
        response.includes("veux") ||
        response.includes("cherche") ||
        response.includes("aimerais") ||
        response.includes("acheter") ||
        response.includes("s il vous plait") ||
        response.includes("svp")
      ) {
        addPoints(10);
        setStage("color");
        setMessage("Très bien ! +10 points");
      } else {
        setMessage(
          "Utilise « Je voudrais... » ou « Je cherche... »."
        );
        return;
      }
    }

    if (stage === "color") {
      if (
        response.includes(normalize(chosenColor)) ||
        response.includes("couleur")
      ) {
        addPoints(5);
        setStage("price");
        setMessage("Excellent ! +5 points");
      } else {
        setMessage(
          `Essaie d'utiliser la couleur « ${chosenColor} ».`
        );
        return;
      }
    }

    if (stage === "price") {
      if (
        response.includes("combien") ||
        response.includes("prix") ||
        response.includes("coute") ||
        response.includes("euros")
      ) {
        addPoints(10);
        setStage("decision");
        setMessage("Formidable ! +10 points");
      } else {
        setMessage(
          "Demande : « Combien ça coûte ? »"
        );
        return;
      }
    }

    setAnswer("");
    setHint(false);
  }

  function purchase() {
    if (!selected || !player) return;

    const alreadyBought = player.cart.some(
      (item) => item.productId === selected.id
    );

    if (alreadyBought) {
      setMessage(
        "Tu as déjà acheté cet article. Choisis-en un autre."
      );
      return;
    }

    if (selected.price > remaining) {
      setMessage(
        "Tu n'as pas assez d'argent pour cet article."
      );
      return;
    }

    updatePlayer((p) => {
      const nextCart = [
        ...p.cart,
        {
          productId: selected.id,
          color: chosenColor,
        },
      ];

      const nextBadges = [...p.badges];

      if (!nextBadges.includes("premier-achat")) {
        nextBadges.push("premier-achat");
      }

      if (
        nextCart.length >= 5 &&
        !nextBadges.includes("fashionista")
      ) {
        nextBadges.push("fashionista");
      }

      return {
        ...p,
        cart: nextCart,
        points: p.points + 15,
        earned: p.earned + 15,
        badges: nextBadges,
      };
    });

    speak("Merci beaucoup ! Au revoir !");
    setMessage("Achat réussi ! +15 points");
    setSelected(null);
    setStage("greeting");
    setAnswer("");
  }

  function surprise() {
    if (!player || player.surpriseUsed) return;

    const rewards = [5, 10, 15, 20];
    const reward =
      rewards[Math.floor(Math.random() * rewards.length)];

    updatePlayer((p) => ({
      ...p,
      bonus: p.bonus + reward,
      surpriseUsed: true,
      points: p.points + 5,
      earned: p.earned + 5,
    }));

    setMessage(
      `Surprise ! Tu as gagné ${reward} € pour tes achats et 5 points !`
    );
  }

  function finishMission() {
    if (!player || !mission || !canCheckout) {
      setMessage(
        "Complète les achats nécessaires avant de terminer."
      );
      return;
    }

    const purchasedItems = [...player.cart];
    const totalSpent = spent;
    const moneyLeft = remaining;

    const completionBonus = 50;
    const budgetBonus =
      missionIndex === 2 && moneyLeft >= 15 ? 25 : 0;

    const earnedThisMission =
      player.earned + completionBonus + budgetBonus;

    const alreadyCompleted = player.history.some(
      (entry) => entry.mission === mission.name
    );

    const newHistoryItem: HistoryItem = {
      mission: mission.name,
      points: earnedThisMission,
      date: new Date().toISOString(),
    };

    updatePlayer((p) => {
      const nextHistory = [
        ...p.history,
        newHistoryItem,
      ];

      const nextCompleted = new Set(
        nextHistory.map((entry) => entry.mission)
      ).size;

      const badges = [...p.badges];

      if (
        nextCompleted >= 3 &&
        !badges.includes("explorateur")
      ) {
        badges.push("explorateur");
      }

      if (
        nextHistory.length >= 5 &&
        !badges.includes("polyglotte")
      ) {
        badges.push("polyglotte");
      }

      return {
        ...p,
        points:
          p.points + completionBonus + budgetBonus,
        earned: earnedThisMission,
        completed: nextCompleted,
        history: nextHistory,
        activeMission: null,
        cart: [],
        bonus: 0,
        surpriseUsed: false,
        badges,
      };
    });

    setReceipt({
      mission: mission.name,
      items: purchasedItems,
      earned: earnedThisMission,
      spent: totalSpent,
      remaining: moneyLeft,
    });

    setSelected(null);
    setMessage(
      alreadyCompleted
        ? "Mission rejouée avec succès !"
        : "Félicitations ! Mission accomplie !"
    );
    setScreen("receipt");
  }

          >
            <div className="flex items-center justify-between border-b border-pink-100 p-5">
              <div>
                <p className="text-xs font-bold uppercase tracking-wider text-[#b63c72]">
                  Dialogue en français
                </p>

                <h2 className="mt-1 text-xl font-black">
                  {selected.shop}
                </h2>
              </div>

              <button
                type="button"
                onClick={() => setSelected(null)}
                className="rounded-full bg-gray-100 px-3 py-2 font-bold text-gray-600"
              >
                ✕
              </button>
            </div>

            <div className="p-6">
              <div className="flex gap-4">
                <Photo
                  product={selected}
                  className="h-32 w-28 rounded-2xl"
                />

                <div className="flex-1">
                  <h3 className="text-xl font-black">
                    {selected.name}
                  </h3>

                  <p className="mt-1 text-sm text-gray-500">
                    {selected.english}
                  </p>

                  <p className="mt-3 text-2xl font-black text-[#b63c72]">
                    {selected.price} €
                  </p>

                  <button
                    type="button"
                    onClick={() =>
                      speak(selected.name)
                    }
                    className="mt-3 rounded-full bg-pink-50 px-4 py-2 text-sm font-bold text-[#b63c72]"
                  >
                    🔊 Écouter
                  </button>
                </div>
              </div>

              <div className="mt-6 rounded-2xl bg-pink-50 p-5">
                <p className="text-xs font-black uppercase tracking-wider text-[#b63c72]">
                  Le vendeur
                </p>

                <p className="mt-2 text-lg font-semibold">
                  {stage === "greeting"
                    ? "Bonjour ! Bienvenue dans notre boutique !"
                    : stage === "request"
                      ? "Comment puis-je vous aider ?"
                      : stage === "color"
                        ? "Quelle couleur préférez-vous ?"
                        : stage === "price"
                          ? `Très bon choix ! ${selected.name} est disponible en ${chosenColor}.`
                          : `Cet article coûte ${selected.price} euros. Voulez-vous l'acheter ?`}
                </p>

                <button
                  type="button"
                  onClick={() =>
                    speak(
                      stage === "greeting"
                        ? "Bonjour ! Bienvenue dans notre boutique !"
                        : stage === "request"
                          ? "Comment puis-je vous aider ?"
                          : stage === "color"
                            ? "Quelle couleur préférez-vous ?"
                            : stage === "price"
                              ? `Très bon choix ! ${selected.name} est disponible en ${chosenColor}.`
                              : `Cet article coûte ${selected.price} euros. Voulez-vous l'acheter ?`
                    )
                  }
                  className="mt-3 rounded-full bg-white px-4 py-2 text-sm font-bold text-[#b63c72]"
                >
                  🔊 Écouter le vendeur
                </button>
              </div>

              {stage !== "decision" ? (
                <>
                  <div className="mt-6">
                    <p className="text-sm font-black text-gray-700">
                      À toi de parler !
                    </p>

                    <p className="mt-2 text-sm text-gray-500">
                      {stage === "greeting"
                        ? "Salue le vendeur en français."
                        : stage === "request"
                          ? `Demande poliment ${selected.name}.`
                          : stage === "color"
                            ? "Dis quelle couleur tu préfères."
                            : "Demande le prix de l'article."}
                    </p>
                  </div>

                  {stage === "color" && (
                    <div className="mt-4">
                      <p className="mb-2 text-sm font-bold">
                        Choisis une couleur :
                      </p>

                      <div className="flex flex-wrap gap-2">
                        {selected.colors.map((color) => (
                          <button
                            key={color}
                            type="button"
                            onClick={() =>
                              setChosenColor(color)
                            }
                            className={`flex items-center gap-2 rounded-full border-2 px-3 py-2 text-sm font-bold ${
                              chosenColor === color
                                ? "border-[#b63c72] bg-pink-50"
                                : "border-gray-200"
                            }`}
                          >
                            <Dot color={color} />
                            {color}
                          </button>
                        ))}
                      </div>
                    </div>
                  )}

                  <input
                    type="text"
                    value={answer}
                    onChange={(event) =>
                      setAnswer(event.target.value)
                    }
                    onKeyDown={(event) => {
                      if (event.key === "Enter") {
                        advance();
                      }
                    }}
                    placeholder="Écris ta réponse en français..."
                    className="mt-4 w-full rounded-xl border-2 border-pink-100 px-4 py-3 outline-none focus:border-pink-400"
                  />

                  <button
                    type="button"
                    onClick={() =>
                      setHint((current) => !current)
                    }
                    className="mt-3 text-sm font-bold text-[#b63c72]"
                  >
                    💡 {hint ? "Masquer l'aide" : "Un indice"}
                  </button>

                  {hint && (
                    <div className="mt-3 rounded-xl bg-yellow-50 p-4 text-sm text-yellow-900">
                      {stage === "greeting"
                        ? "Bonjour !"
                        : stage === "request"
                          ? `Je voudrais acheter ${selected.name}, s'il vous plaît.`
                          : stage === "color"
                            ? `Je préfère la couleur ${chosenColor}.`
                            : "Combien ça coûte ?"}
                    </div>
                  )}

                  {message && (
                    <p className="mt-4 rounded-xl bg-blue-50 p-3 text-sm font-semibold text-blue-800">
                      {message}
                    </p>
                  )}

                  <button
                    type="button"
                    onClick={advance}
                    disabled={!answer.trim()}
                    className="mt-5 w-full rounded-2xl bg-[#b63c72] px-5 py-4 font-black text-white hover:bg-[#982e5c] disabled:bg-gray-300"
                  >
                    Envoyer ma réponse →
                  </button>
                </>
              ) : (
                <div className="mt-6">
                  <h3 className="text-xl font-black">
                    Veux-tu acheter cet article ?
                  </h3>

                  <p className="mt-3 text-sm text-gray-600">
                    Prix : {selected.price} € — Il te
                    reste {remaining} €.
                  </p>

                  {message && (
                    <p className="mt-4 rounded-xl bg-blue-50 p-3 text-sm text-blue-800">
                      {message}
                    </p>
                  )}

                  <div className="mt-6 grid grid-cols-2 gap-3">
                    <button
                      type="button"
                      onClick={purchase}
                      disabled={selected.price > remaining}
                      className="rounded-2xl bg-green-600 px-4 py-4 font-black text-white hover:bg-green-700 disabled:bg-gray-300"
                    >
                      Oui, je l'achète !
                    </button>

                    <button
                      type="button"
                      onClick={() => setSelected(null)}
                      className="rounded-2xl bg-gray-100 px-4 py-4 font-black text-gray-700"
                    >
                      Non, merci !
                    </button>
                  </div>
                </div>
              )}
            </div>
          </div>
        </div>
      )}

      <footer className="border-t border-pink-100 bg-white px-4 py-6 text-center text-xs text-gray-500">
        Mission Paris : Le Shopping Secret 🇫🇷
        <span className="mx-2">•</span>
        Apprendre le français en s'amusant
      </footer>
    </main>
  );
}
